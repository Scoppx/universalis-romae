# Universalis Romae — Tappa 3: Jev guida i popoli — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** I popoli non giocati prendono le decisioni strategiche (guerra, obiettivo degli eserciti, accettare una pace) chiedendole a Jev, in modo asincrono e senza fermare il gioco; un pannello "Consiglio" mostra scelte e probabilità; il bench confronta Jev e regole e può rigiocare una partita dal registro.

**Architecture:** Il server fa da proxy a TypeSafe (`POST /api/jev`), la chiave resta in Node. `ai/questions.js` (puro) costruisce stato e domande dai fatti calcolati dal codice e interpreta le risposte. `ai/jev.js` è un'IA asincrona: accoda le domande, applica le risposte arrivate come comandi al primo giorno utile, ricade sulle regole se Jev non risponde o risponde tardi, e registra ogni decisione. Il lavoro di fino (eserciti, reclute, proposte di pace) resta alle regole. Un nuovo campo `obiettivo` per popolo, impostato da comando, orienta gli eserciti.

**Tech Stack:** come tappe 1–2. API TypeSafe: `POST https://api.typesafe.ai/v1/systemone`, `Authorization: Bearer <key>`, corpo `{ model: 'jev-latest', state, questions }`; risposte `answers[q]` con `choice`/`probabilities`/`confidence` (choice) o `noul` (noul); errori 401/422/429/529.

**Spec:** `docs/superpowers/specs/2026-10-12-universalis-romae-design.md` (sezione 5, IA e Jev).

**Esecuzione:** inline (i compiti condividono `questions.js`, `jev.js` e `rules.js`).

## Global Constraints

- `TYPESAFE_API_KEY` solo lato Node (server e bench), mai nel browser. Server su `127.0.0.1`.
- La simulazione resta deterministica: Jev entra solo come comandi; il bench aspetta le risposte a fine giorno (`flush`) e le applica il giorno dopo, in ordine di richiesta.
- Interpretazione (spec): guerra e obiettivo prendono l'opzione più probabile; se la migliore è sotto 0,5 e il distacco dalla seconda è sotto 0,15 si sceglie l'opzione prudente (`nessuna guerra`, `difendi`). Pace: si accetta se `noul ≥ 0,6`.
- Le soglie e i vincoli restano in codice: Jev sceglie solo tra opzioni già legali (niente guerra sopra la soglia d'infamia o senza pretesto con stabilità negativa).
- Opzioni con significato diverso, mai scelte numeriche fini (regola dai progetti Jev precedenti).
- Risposta arrivata più di 60 giorni dopo la domanda: scartata, decide la regola. Al massimo 4 richieste in parallelo, coda di 40.
- Revisione strategica ogni 91 giorni sfalsata per popolo, solo se c'è qualcosa da decidere (spec, sezione 5 corretta).
- Modalità IA: `regole`, `jev`, `misto` (popoli in posizione pari in `game.ordine` con Jev, dispari con le regole).
- Commit alla fine di ogni task con `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

1. Jev lento o fermo mentre il gioco corre: il gioco non aspetta mai, le risposte vecchie oltre 60 giorni vengono scartate, la coda non cresce senza limite (test nel task 4).
2. Risposta di Jev malformata o con un'opzione inesistente: si decide con le regole, niente eccezioni (test nei task 3 e 4).
3. Risposta che arriva quando la situazione è cambiata (già in guerra, proposta scaduta, popolo eliminato): nessun comando assurdo (test nel task 4).
4. Server senza chiave o TypeSafe in errore (401/429/529): il gioco funziona con le regole, l'interfaccia lo dice una volta (task 2 e prova di fumo nel task 6).
5. Rigiocare una partita dal registro dà esattamente lo stesso stato finale (test nel task 4).

// Lavoro di fino, uguale per regole e Jev: eserciti ogni 7 giorni (sfalsati) e reclute il primo del mese.
export function routine(game, id) {
  const cmds = [];
  if ((game.day + hashId(id)) % 7 === 0) cmds.push(...armyOrders(game, id));
  if (isFirstOfMonth(game.day)) cmds.push(...recruitment(game, id));
  return cmds;
}

export function rulesController(game, id) {
  const cmds = [...answerProposals(game, id)];
  if (isReviewDay(game, id)) cmds.push(...strategy(game, id));
  cmds.push(...routine(game, id));
  return cmds;
}

export function strategy(game, id) {
  const inWar = Object.values(game.guerre).some(w => sideOf(w, id));
  return inWar ? peaceProposals(game, id) : warDecision(game, id);
}

// Vicini che il codice permette di attaccare: sotto la soglia d'infamia, e con pretesto o stabilità non negativa.
export function legalWarTargets(game, id) {
  const n = game.nazioni[id];
  if (n.infamia >= game.balance.infamia.soglia) return [];
  return warCandidates(game, id).filter(c => hasPretext(game, id, c.id) || n.stabilita >= 0);
}

export function warDecision(game, id) {
  const best = legalWarTargets(game, id)[0];
  if (!best || best.rapporto < game.balance.ia.rapportoGuerra) return [];
  return [{ tipo: 'dichiara_guerra', nazione: id, contro: best.id }];
}

export function peaceProposals(game, id) {
  const ia = game.balance.ia;
  const cmds = [];
  for (const war of Object.values(game.guerre).filter(w => sideOf(w, id))) {
    const side = sideOf(war, id);
    if (war[side][0] !== id) continue; // decide il capofila
    if (Object.values(game.proposte).some(p => p.guerra === war.id && p.da === id)) continue;
    const enemySide = side === 'attaccanti' ? 'difensori' : 'attaccanti';
    const enemyLeader = enemiesOf(war, side)[0];
    const s = scoreFor(game, war, id);
    const byTax = (a, b) => b.tasse - a.tasse || a.id.localeCompare(b.id);
    if (s >= ia.pacePunteggio) {
      const occupied = Object.values(game.province)
        .filter(p => war[enemySide].includes(p.owner) && war[side].includes(p.controller)).sort(byTax);
      const scelte = [];
      for (const p of occupied) if (peaceCost(game, war, id, { province: [...scelte, p.id] }) <= s) scelte.push(p.id);
      if (scelte.length) cmds.push({ tipo: 'proponi_pace', nazione: id, guerra: war.id, termini: { ricevente: id, province: scelte } });
    } else if (s <= ia.resaPunteggio) {
      const lost = Object.values(game.province)
        .filter(p => war[side].includes(p.owner) && war[enemySide].includes(p.controller)).sort(byTax);
      const scelte = [];
      for (const p of lost) if (peaceCost(game, war, enemyLeader, { province: [...scelte, p.id] }) <= -s) scelte.push(p.id);
      cmds.push({ tipo: 'proponi_pace', nazione: id, guerra: war.id, termini: { ricevente: enemyLeader, province: scelte } });
    } else if (game.day - war.inizio >= ia.paceBiancaAnni * 365 && Math.abs(s) < 10) {
      cmds.push({ tipo: 'proponi_pace', nazione: id, guerra: war.id, termini: { ricevente: id, province: [] } });
    }
  }
  return cmds;
}

```js
// Chiamata a TypeSafe System One (Jev). Usata dal proxy del server e dal bench; la chiave resta in Node.
export const JEV_URL = 'https://api.typesafe.ai/v1/systemone';
export const JEV_MODEL = 'jev-latest';

export function createAsk({ key, fetchImpl = fetch, timeoutMs = 15000, retryMs = 1000 }) {
  async function once({ state, questions }) {
    const res = await fetchImpl(JEV_URL, {
      method: 'POST',
      headers: { Authorization: `Bearer ${key}`, 'Content-Type': 'application/json' },
      body: JSON.stringify({ model: JEV_MODEL, state, questions }),
      signal: AbortSignal.timeout(timeoutMs)
    });
    const text = await res.text();
    return { status: res.status, ok: res.ok, text };
  }
  return async function ask(body) {
    let r = await once(body);
    // 429 e 529: TypeSafe chiede di riprovare dopo un attimo
    if (r.status === 429 || r.status === 529) {
      await new Promise(resolve => setTimeout(resolve, retryMs));
      r = await once(body);
    }
    if (!r.ok) throw new Error(`Jev ${r.status}: ${r.text.slice(0, 200)}`);
    return JSON.parse(r.text);
  };
}
```

async function readBody(req, limit) {
  let size = 0;
  const chunks = [];
  for await (const chunk of req) {
    size += chunk.length;
    if (size > limit) return null;
    chunks.push(chunk);
  }
  return Buffer.concat(chunks).toString('utf8');
}

function sendJson(res, status, body) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store' });
  res.end(JSON.stringify(body));
}

async function proxyJev(req, res, ask) {
  if (!ask) return sendJson(res, 503, { errore: 'Jev non disponibile: avvia il server con TYPESAFE_API_KEY' });
  const text = await readBody(req, 256 * 1024);
  if (text === null) return sendJson(res, 413, { errore: 'richiesta troppo grande' });
  let body;
  try { body = JSON.parse(text); } catch { return sendJson(res, 400, { errore: 'JSON non valido' }); }
  if (!body || body.state === undefined || typeof body.questions !== 'object' || body.questions === null) {
    return sendJson(res, 400, { errore: 'servono state e questions' });
  }
  try {
    sendJson(res, 200, await ask({ state: body.state, questions: body.questions }));
  } catch (e) {
    sendJson(res, 502, { errore: String(e.message || e) });
  }
}

export function createServer(root = ROOT, { ask = null } = {}) {
  return http.createServer(async (req, res) => {
    const pathname = req.url.split('?')[0];
    if (pathname === '/api/jev') {
      if (req.method === 'GET') return sendJson(res, 200, { disponibile: Boolean(ask) });
      if (req.method === 'POST') return proxyJev(req, res, ask);
      res.writeHead(405).end();
      return;
    }
    if (req.method !== 'GET') { res.writeHead(405).end(); return; }
    let rel;
    try {
      const p = new URL(req.url, 'http://localhost').pathname;
      rel = p === '/' ? 'index.html' : decodeURIComponent(p.slice(1));
    } catch {
      res.writeHead(400).end();
      return;
    }
    const file = path.join(root, path.normalize(rel));
    if (!file.startsWith(root + path.sep)) { res.writeHead(403).end(); return; }
    try {
      const body = await readFile(file);

import { legalWarTargets, enemiesOfNation, power } from './rules.js';
import { distancesFrom } from '../sim/armies.js';
import { armiesOf } from '../sim/state.js';
import { sideOf, enemiesOf, scoreFor, hasPretext, peaceCost } from '../sim/war.js';
import { formatDate } from '../sim/calendar.js';

export const SOGLIA_SI = 0.6;
const round2 = x => Math.round(x * 100) / 100;
const nome = (game, id) => game.nazioni[id]?.nome ?? id;

function statoPopolo(game, id, carattere) {
  const n = game.nazioni[id];
  return {
    data: formatDate(game.day),
    popolo: {
      nome: n.nome, carattere, governo: n.governo, stabilita: n.stabilita,
      infamia: Math.round(n.infamia), soglia_infamia: game.balance.infamia.soglia,
      tesoro: Math.round(n.tesoro), forza: Math.round(power(game, id))
    }
  };
}

function guerreDi(game, id) {
  return Object.values(game.guerre).filter(w => sideOf(w, id)).map(w => {
    const side = sideOf(w, id), nemici = enemiesOf(w, side);
    const prov = Object.values(game.province);
    return {
      contro: nemici.map(n => nome(game, n)).join(', '),
      punteggio: Math.round(scoreFor(game, w, id)),
      durata_anni: Math.round((game.day - w.inizio) / 36.5) / 10,
      nostre_province_occupate: prov.filter(p => p.owner === id && nemici.includes(p.controller)).length,
      loro_province_occupate_da_noi: prov.filter(p => nemici.includes(p.owner) && p.controller === id).length
    };
  });

// Obiettivi possibili per l'esercito principale, ciascuno con un significato diverso.
function objectiveOptions(game, id) {
  const armies = armiesOf(game, id);
  if (!armies.length) return {};
  const main = armies.reduce((a, b) => (b.fanteria + b.cavalleria > a.fanteria + a.cavalleria ? b : a));
  const { dist } = distancesFrom(game, id, main.prov);
  const enemies = enemiesOfNation(game, id);
  const n = game.nazioni[id];
  const P = pid => game.province[pid].nome;
  const byDist = (a, b) => dist[a.id] - dist[b.id] || (a.id < b.id ? -1 : 1);
  const raggiungibili = Object.values(game.province).filter(p => dist[p.id] !== undefined);
  const opts = {};
  opts.difendi = { prov: n.capitale, testo: `Difendere la capitale ${P(n.capitale)} e aspettare il nemico in casa.` };
  const perse = raggiungibili.filter(p => p.owner === id && enemies.includes(p.controller)).sort(byDist);
  if (perse.length) opts.libera = { prov: perse[0].id, testo: `Liberare ${P(perse[0].id)}, nostra provincia occupata da ${nome(game, perse[0].controller)}.` };
  const nemiche = raggiungibili.filter(p => enemies.includes(p.owner) && enemies.includes(p.controller)).sort(byDist);
  if (nemiche.length) {
    const v = nemiche[0];
    opts.vicina = { prov: v.id, testo: `Assediare ${P(v.id)} di ${nome(game, v.owner)}, la provincia nemica più vicina (${dist[v.id]} giorni di marcia).` };
    const r = [...nemiche].sort((a, b) => b.tasse - a.tasse || byDist(a, b))[0];
    if (r.id !== v.id) opts.ricca = { prov: r.id, testo: `Assediare ${P(r.id)}, la provincia nemica più ricca (tasse ${r.tasse}, ${dist[r.id]} giorni di marcia).` };
    const cap = nemiche.find(p => enemies.some(e => game.nazioni[e].capitale === p.id));
    if (cap && cap.id !== v.id && cap.id !== r.id) {
      opts.capitale = { prov: cap.id, testo: `Marciare sulla capitale nemica ${P(cap.id)} (${dist[cap.id]} giorni): se cade, la guerra è quasi vinta.` };
    }
  }
  return opts;
}

export function buildReview(game, id, carattere) {
  const inWar = Object.values(game.guerre).some(w => sideOf(w, id));
  const state = statoPopolo(game, id, carattere);
  const questions = {}, meta = {};
  if (!inWar) {
    const legal = legalWarTargets(game, id);
    if (!legal.length) return null;
    state.vicini_attaccabili = legal.map(c => ({
      nome: nome(game, c.id), rapporto_forze: round2(c.rapporto), pretesto: hasPretext(game, id, c.id)
    }));
    const criteria = { 'nessuna guerra': 'Restare in pace per ora: nessun vicino vale il rischio di una guerra adesso.' };
    const mappa = { 'nessuna guerra': null };
    for (const c of legal) {
      const N = nome(game, c.id);
      criteria[N] = `Dichiarare guerra a ${N}: le nostre forze sono ${round2(c.rapporto)} volte le sue` +
        (hasPretext(game, id, c.id) ? ', e abbiamo un pretesto.' : '; senza pretesto perdiamo stabilità.');
      mappa[N] = c.id;
    }
    questions.guerra = {
      type: 'choice',
      instructions: 'Sei il consiglio di `popolo.nome` e decidi secondo `popolo.carattere`. Guardando `vicini_attaccabili`, il rapporto di forze e la situazione interna, cosa fai adesso?',
      criteria
    };
    meta.guerra = { tipo: 'guerra', mappa, prudente: 'nessuna guerra' };
  } else {
    const opts = objectiveOptions(game, id);
    if (Object.keys(opts).length >= 2) {
      state.guerre = guerreDi(game, id);
      questions.obiettivo = {
        type: 'choice',
        instructions: 'Sei il consiglio di `popolo.nome` e decidi secondo `popolo.carattere`. Guardando `guerre`, dove mandi il grosso dell\'esercito nella prossima stagione?',
        criteria: Object.fromEntries(Object.entries(opts).map(([k, o]) => [k, o.testo]))
      };
      meta.obiettivo = { tipo: 'obiettivo', mappa: Object.fromEntries(Object.entries(opts).map(([k, o]) => [k, o.prov])), prudente: 'difendi' };
    }
  }
  return Object.keys(questions).length ? { state, questions, meta } : null;
}

export function buildPeaceQuestion(game, propId, carattere) {
  const p = game.proposte[propId], war = game.guerre[p.guerra], id = p.a;
  const state = statoPopolo(game, id, carattere);
  const cediamo = p.termini.ricevente !== id;
  state.guerra = {
    contro: nome(game, p.da), punteggio: Math.round(scoreFor(game, war, id)),
    durata_anni: Math.round((game.day - war.inizio) / 36.5) / 10
  };
  state.proposta = {
    da: nome(game, p.da),
    tipo: !p.termini.province.length ? 'pace bianca' : cediamo ? 'ci chiede province' : 'ci offre province',
    province: p.termini.province.map(pid => ({ nome: game.province[pid].nome, tasse: game.province[pid].tasse })),
    costo_in_punteggio: Math.round(peaceCost(game, war, p.termini.ricevente, p.termini))
  };
  return {
    state,
    questions: {
      pace: {
        type: 'noul',
        instructions: 'Il consiglio di `popolo.nome`, che decide secondo `popolo.carattere`, accetta `proposta` per chiudere `guerra`?'
      }
    },
    meta: { pace: { tipo: 'pace', proposta: propId } }
  };
}

export function interpretChoice(answer, meta) {
  const probs = answer?.probabilities;
  if (!probs || typeof probs !== 'object') return null;
  const sorted = Object.entries(probs).filter(([, v]) => typeof v === 'number').sort((a, b) => b[1] - a[1]);
  if (!sorted.length || !Object.hasOwn(meta.mappa, sorted[0][0])) return null;
  const [top, p1] = sorted[0], p2 = sorted[1]?.[1] ?? 0;
  const incerto = p1 < 0.5 && p1 - p2 < 0.15;
  const scelta = incerto ? meta.prudente : top;
  return { scelta, valore: meta.mappa[scelta], incerto, probabilita: probs };
}

export function interpretNoul(answer) {
  const v = answer?.noul;
  if (typeof v !== 'number' || !Number.isFinite(v)) return null;
  return { accetta: v >= SOGLIA_SI, probabilita: v };
}

  let seq = 0;

  function pump() {
    while (inVolo.size < maxParallel && coda.length) {
      const job = coda.shift();
      const p = Promise.resolve()
        .then(() => ask(job.req))
        .then(res => { pronte.push({ job, res }); }, err => { pronte.push({ job, err: String(err?.message || err) }); })
        .finally(() => { inVolo.delete(p); pump(); });
      inVolo.add(p);
    }
  }

  function enqueue(job) {
    job.seq = seq++;
    if (coda.length >= maxCoda) pronte.push({ job: coda.shift(), err: 'coda piena' });
    coda.push(job);
    pump();
  }

  function applica(game, { job, res, err }) {
    const id = job.nazione;
    const scaduta = game.day - job.giorno > scadenza;
    const usabile = !err && !scaduta && game.nazioni[id].vivo;
    const record = { seq: job.seq, giorno: job.giorno, applicato: game.day, nazione: id, domande: [], comandi: [], usage: res?.usage ?? null, errore: err ?? (scaduta ? 'risposta oltre i 60 giorni' : null) };
    for (const [q, meta] of Object.entries(job.meta)) {
      const ans = res?.answers?.[q];
      if (meta.tipo === 'pace') {
        if (!game.proposte[meta.proposta]) { record.domande.push({ domanda: 'pace', scelta: null, fonte: 'superata', probabilita: null }); continue; }
        const v = usabile ? interpretNoul(ans) : null;
        const accetta = v ? v.accetta : answerProposals(game, id).find(c => c.proposta === meta.proposta).accetta;
        record.comandi.push({ tipo: 'rispondi', nazione: id, proposta: meta.proposta, accetta });
        record.domande.push({ domanda: 'pace', scelta: accetta ? 'accetta' : 'rifiuta', fonte: v ? 'jev' : 'regola', probabilita: v ? v.probabilita : null });
        continue;
      }
      if (meta.tipo === 'guerra' && inGuerra(game, id)) {
        record.domande.push({ domanda: 'guerra', scelta: null, fonte: 'superata', probabilita: null });
        continue;
      }
      const v = usabile ? interpretChoice(ans, meta) : null;
      if (meta.tipo === 'guerra') {
        if (v) { if (v.valore) record.comandi.push({ tipo: 'dichiara_guerra', nazione: id, contro: v.valore }); }
        else record.comandi.push(...warDecision(game, id));
      } else if (meta.tipo === 'obiettivo' && v) {
        record.comandi.push({ tipo: 'obiettivo', nazione: id, provincia: v.valore });
      }
      record.domande.push({ domanda: meta.tipo, scelta: v ? v.scelta : 'regola', fonte: v ? (v.incerto ? 'prudenza' : 'jev') : 'regola', probabilita: v ? v.probabilita : null });
    }
    onRecord(record);
    return record.comandi;
  }

  function controller(game, id) {
    const cmds = [];
    const mie = pronte.filter(r => r.job.nazione === id).sort((a, b) => a.job.seq - b.job.seq);
    for (const r of mie) {
      pronte.splice(pronte.indexOf(r), 1);
      cmds.push(...applica(game, r));
    }
    for (const p of Object.values(game.proposte)) {
      if (p.a !== id || chieste.has(p.id)) continue;
      chieste.add(p.id);
      const q = buildPeaceQuestion(game, p.id, caratteri[id] ?? '');
      enqueue({ nazione: id, giorno: game.day, req: { state: q.state, questions: q.questions }, meta: q.meta });
    }
    if (isReviewDay(game, id)) {
      if (inGuerra(game, id)) cmds.push(...peaceProposals(game, id));
      const q = buildReview(game, id, caratteri[id] ?? '');
      if (q) enqueue({ nazione: id, giorno: game.day, req: { state: q.state, questions: q.questions }, meta: q.meta });
    }
    cmds.push(...routine(game, id));
    return cmds;
  }

  async function flush() {
    while (inVolo.size || coda.length) await Promise.all([...inVolo]);
  }

  return { controller, flush, inAttesa: () => coda.length + inVolo.size };
}

// Rigioca una partita dal registro: stessi comandi negli stessi giorni, stesso lavoro di fino.
export function createReplayAi(records) {
  const perGiorno = new Map();
  for (const r of [...records].sort((a, b) => a.seq - b.seq)) {
    const k = `${r.applicato}|${r.nazione}`;
    if (!perGiorno.has(k)) perGiorno.set(k, []);
    perGiorno.get(k).push(...r.comandi);
  }
  function controller(game, id) {
    const cmds = [...(perGiorno.get(`${game.day}|${id}`) ?? [])];
    if (isReviewDay(game, id) && inGuerra(game, id)) cmds.push(...peaceProposals(game, id));
    cmds.push(...routine(game, id));
    return cmds;
  }
  return { controller };
}
